# Scenarios for compression-bench.sh. One per line:
#   name | startup arguments | runtime settings (name value ...)
# The script always adds: --save '' --protected-mode no --maxmemory-policy allkeys-lfu
# Each group changes one thing from its base. The run order is the file order.

# 1. Baseline: no compression.
baseline-off       | --io-threads 12 |
